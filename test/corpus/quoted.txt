================================================================================
Semicolon after a quoted name
================================================================================

"block";{}
"param" ;:i=1
'simple';=text

--------------------------------------------------------------------------------

(document
  (block name: (string))
  (parameter name: (string) type: (type) value: (raw_value))
  (parameter name: (string) value: (raw_value)))

================================================================================
Quoted parameter types
================================================================================

a:"i"=5
b:'p3';=1,2,3
c:"t"[]=["x"]
d:"~i"=6
e:"""i64"""=7

--------------------------------------------------------------------------------

(document
  (parameter name: (identifier) type: (type (string)) value: (raw_value))
  (parameter name: (identifier) type: (type (string)) value: (raw_value))
  (parameter name: (identifier) type: (type (string)) value: (array (string)))
  (parameter name: (identifier) type: (type (string)) value: (raw_value))
  (parameter name: (identifier) type: (type (string)) value: (raw_value)))
